     1→'use client';
     2→
     3→import { useEffect, useMemo, useRef, useState } from 'react';
     4→import { useQuery } from '@tanstack/react-query';
     5→import { toast } from 'sonner';
     6→import { ArrowLeft, ArrowRight, Check, Heart, Leaf, Recycle } from 'lucide-react';
     7→import type { ProvinceId, ViewProps } from '@/lib/types';
     8→import { fetchFarmers, fetchProduct, fetchProducts } from '@/lib/api';
     9→import { useLang } from '@/lib/stores/lang';
    10→import { useRouterStore } from '@/lib/stores/router';
    11→import { useCartStore } from '@/lib/stores/cart';
    12→import { useWishlistStore } from '@/lib/stores/wishlist';
    13→import { useRecentStore } from '@/lib/stores/recent';
    14→import { useMounted } from '@/lib/hooks';
    15→import { useSeo } from '@/lib/useSeo';
    16→import { getCategory } from '@/lib/data/categories';
    17→import { getProvince, provinceName } from '@/lib/data/provinces';
    18→import { EmptyState } from '@/components/shared/EmptyState';
    19→import { ProductCard, formatPrice } from '@/components/shared/ProductCard';
    20→import { QuantityStepper } from '@/components/shared/QuantityStepper';
    21→import { RatingStars } from '@/components/shared/RatingStars';
    22→import { Reveal } from '@/components/shared/Reveal';
    23→import { SectionHeading } from '@/components/shared/SectionHeading';
    24→import { ShareButtons } from '@/components/shared/ShareButtons';
    25→import { SmartImage } from '@/components/shared/SmartImage';
    26→import { OriginChain } from '@/components/product/OriginChain';
    27→import { ProductReviews } from '@/components/product/ProductReviews';
    28→import {
    29→  Accordion,
    30→  AccordionContent,
    31→  AccordionItem,
    32→  AccordionTrigger,
    33→} from '@/components/ui/accordion';
    34→import { cn } from '@/lib/utils';
    35→
    36→// ─── ProductView — gallery, craft details, origin story & related harvests ────
    37→
    38→export default function ProductView({ view }: ViewProps) {
    39→  const { t, lang } = useLang();
    40→  const navigate = useRouterStore((s) => s.navigate);
    41→  const add = useCartStore((s) => s.add);
    42→  const toggleWishlist = useWishlistStore((s) => s.toggle);
    43→  const wishlistIds = useWishlistStore((s) => s.ids);
    44→  const recordRecent = useRecentStore((s) => s.record);
    45→  const recentSlugs = useRecentStore((s) => s.slugs);
    46→  const mounted = useMounted();
    47→
    48→  const slug = view.name === 'product' ? view.slug : '';
    49→
    50→  // ── Data ────────────────────────────────────────────────────────────────────
    51→  const {
    52→    data: product,
    53→    isError,
    54→  } = useQuery({
    55→    queryKey: ['product', slug],
    56→    queryFn: () => fetchProduct(slug),
    57→    enabled: slug.length > 0,
    58→  });
    59→  const { data: farmers } = useQuery({ queryKey: ['farmers'], queryFn: fetchFarmers });
    60→  const { data: allProducts } = useQuery({ queryKey: ['products'], queryFn: fetchProducts });
    61→
    62→  const farmer = useMemo(
    63→    () => farmers?.find((f) => f.id === product?.farmerId),
    64→    [farmers, product?.farmerId],
    65→  );
    66→  const prov = useMemo(() => (product ? getProvince(product.province) : undefined), [product]);
    67→
    68→  // ── Local state ─────────────────────────────────────────────────────────────
    69→  const [activeImage, setActiveImage] = useState(0);
    70→  const [selectedSize, setSelectedSize] = useState('');
    71→  const [qty, setQty] = useState(1);
    72→  const [justAdded, setJustAdded] = useState(false);
    73→  const timer = useRef<ReturnType<typeof setTimeout> | null>(null);
    74→
    75→  // Selection state initialises from the loaded product; the keyed view wrapper remounts
    76→  // this component on navigation, so no reset effect is required.
    77→
    78→  // Clear the “Added ✓” timer on unmount.
    79→  useEffect(() => () => {
    80→    if (timer.current) clearTimeout(timer.current);
    81→  }, []);
    82→
    83→  // Track this product in the recently-viewed history — only once it has
    84→  // actually loaded, so failed/unknown slugs never pollute the history.
    85→  useEffect(() => {
    86→    if (product) recordRecent(product.slug);
    87→  }, [product?.id, recordRecent]);
    88→
    89→  // ── Gallery (product → farm → province) ─────────────────────────────────────
    90→  const gallery = useMemo(() => {
    91→    if (!product) return [];
    92→    const images: { src: string; alt: string }[] = [
    93→      { src: product.image, alt: product.name },
    94→    ];
    95→    if (farmer?.farmImage) images.push({ src: farmer.farmImage, alt: `${farmer.name} — the farm` });
    96→    if (prov) images.push({ src: prov.image, alt: `${prov.name}, Cambodia — ${prov.tagline}` });
    97→    return images;
    98→  }, [product, farmer, prov]);
    99→
   100→  // ── Related products (same category, topped up with same province) ──────────
   101→  const related = useMemo(() => {
   102→    if (!product || !allProducts) return [];
   103→    const sameCategory = allProducts.filter(
   104→      (p) => p.id !== product.id && p.category === product.category,
   105→    );
   106→    const sameProvince = allProducts.filter(
   107→      (p) =>
   108→        p.id !== product.id &&
   109→        p.category !== product.category &&
   110→        p.province === product.province,
   111→    );
   112→    return [...sameCategory, ...sameProvince].slice(0, 4);
   113→  }, [product, allProducts]);
   114→
   115→  // ── Recently viewed (history strip, excludes the current product) ───────────
   116→  const recentProducts = useMemo(() => {
   117→    if (!product || !allProducts) return [];
   118→    return recentSlugs
   119→      .filter((s) => s !== product.slug)
   120→      .map((s) => allProducts.find((p) => p.slug === s))
   121→      .filter((p): p is NonNullable<typeof p> => Boolean(p))
   122→      .slice(0, 4);
   123→  }, [product, allProducts, recentSlugs]);
   124→
   125→  // SEO — refines the base title once the product resolves (hook is called
   126→  // unconditionally to keep the hook order stable across loading states).
   127→  useSeo(
   128→    product ? `${product.name} — ${product.farmerName} | Sovann Farm` : undefined,
   129→    product
   130→      ? `${product.description} Grown in ${provinceName(product.province)}, Cambodia.`
   131→      : undefined,
   132→  );
   133→
   134→  // ── Loading / not found ─────────────────────────────────────────────────────
   135→  if (product === undefined && !isError) {
   136→    return <ProductViewSkeleton />;
   137→  }
   138→
   139→  // ── Not found ───────────────────────────────────────────────────────────────
   140→  if (!product) {
   141→    return (
   142→      <div className="container-editorial py-24">
   143→        <EmptyState
   144→          title={t('shop.empty')}
   145→          description={t('shop.emptyDesc')}
   146→          action={
   147→            <button
   148→              type="button"
   149→              onClick={() => navigate({ name: 'shop' })}
   150→              className="btn-primary"
   151→            >
   152→              {t('product.shop')}
   153→            </button>
   154→          }
   155→        />
   156→      </div>
   157→    );
   158→  }
   159→
   160→  // ── Derived content ─────────────────────────────────────────────────────────
   161→  const name = lang === 'kh' && product.nameKh ? product.nameKh : product.name;
   162→  const categoryData = getCategory(product.category);
   163→  const categoryNameValue = categoryData
   164→    ? lang === 'kh'
   165→      ? categoryData.nameKh
   166→      : categoryData.name
   167→    : null;
   168→  const provinceLabel =
   169→    product.province === 'multi'
   170→      ? lang === 'kh'
   171→        ? 'ខេត្តច្រើន'
   172→        : 'Multiple Provinces'
   173→      : lang === 'kh'
   174→        ? (prov?.nameKh ?? provinceName(product.province))
   175→        : provinceName(product.province);
   176→
   177→  const sizeOptions = product.sizes;
   178→  const selectedSizeObj =
   179→    sizeOptions.find((s) => s.label === selectedSize) ?? sizeOptions[0];
   180→  const activeSrc = gallery[Math.min(activeImage, Math.max(gallery.length - 1, 0))]?.src ?? product.image;
   181→  const activeAlt =
   182→    gallery[Math.min(activeImage, Math.max(gallery.length - 1, 0))]?.alt ?? product.name;
   183→
   184→  const wished = wishlistIds.includes(product.id);
   185→  const farmerSlug =
   186→    farmer?.slug ??
   187→    (product.farmerId && product.farmerId.startsWith('f-') ? product.farmerId.slice(2) : null);
   188→  const localizedProvince = (id: ProvinceId) =>
   189→    lang === 'kh' ? (getProvince(id)?.nameKh ?? provinceName(id)) : provinceName(id);
   190→
   191→  const handleAdd = () => {
   192→    if (product.stock <= 0) return;
   193→    add(product.id, selectedSizeObj?.label ?? '', qty);
   194→    setJustAdded(true);
   195→    if (timer.current) clearTimeout(timer.current);
   196→    timer.current = setTimeout(() => setJustAdded(false), 1600);
   197→    toast.success(`${name} — ${t('common.added')}`);
   198→  };
   199→
   200→  const handleWishlist = () => {
   201→    toggleWishlist(product.id);
   202→  };
   203→
   204→  const goFarmer = () => {
   205→    if (farmerSlug) navigate({ name: 'farmer', slug: farmerSlug });
   206→  };
   207→
   208→  const accordionTrigger =
   209→    'rounded-none py-5 text-[11px] font-bold uppercase tracking-[0.26em] text-charcoal hover:no-underline hover:text-forest';
   210→
   211→  return (
   212→    <div>
   213→      {/* ── Breadcrumb + back ──────────────────────────────────────────────── */}
   214→      <div className="container-editorial pt-10 md:pt-14">
   215→        <button
   216→          type="button"
   217→          onClick={() => navigate({ name: 'shop' })}
   218→          className="flex cursor-pointer items-center gap-2 text-[11px] font-bold uppercase tracking-[0.22em] text-stone transition-colors hover:text-forest focus-visible:outline-2 focus-visible:outline-gold"
   219→        >
   220→          <ArrowLeft className="h-3.5 w-3.5" strokeWidth={1.5} aria-hidden="true" />
   221→          {t('product.shop')}
   222→        </button>
   223→        <nav
   224→          aria-label="Breadcrumb"
   225→          className="mt-4 flex flex-wrap items-center gap-2 text-[11px] font-semibold uppercase tracking-[0.22em]"
   226→        >
   227→          <button
   228→            type="button"
   229→            onClick={() => navigate({ name: 'shop' })}
   230→            className="cursor-pointer text-stone transition-colors hover:text-forest focus-visible:outline-2 focus-visible:outline-gold"
   231→          >
   232→            {t('product.shop')}
   233→          </button>
   234→          {categoryNameValue && (
   235→            <>
   236→              <span className="text-charcoal/25" aria-hidden="true">/</span>
   237→              <button
   238→                type="button"
   239→                onClick={() => navigate({ name: 'shop', category: product.category })}
   240→                className="cursor-pointer text-stone transition-colors hover:text-forest focus-visible:outline-2 focus-visible:outline-gold"
   241→              >
   242→                {categoryNameValue}
   243→              </button>
   244→            </>
   245→          )}
   246→          <span className="text-charcoal/25" aria-hidden="true">/</span>
   247→          <span className="text-charcoal" aria-current="page">{name}</span>
   248→        </nav>
   249→
   250→        {/* ── Main grid ──────────────────────────────────────────────────────── */}
   251→        <div className="grid gap-12 pb-20 pt-8 lg:grid-cols-2 lg:gap-16">
   252→          {/* LEFT — gallery */}
   253→          <div>
   254→            <div className="group relative overflow-hidden border border-charcoal/10 bg-parchment">
   255→              <SmartImage
   256→                key={activeSrc}
   257→                src={activeSrc}
   258→                alt={activeAlt}
   259→                ratio="portrait"
   260→                priority
   261→                imgClassName="group-hover:scale-105"
   262→              />
   263→            </div>
   264→            <div className="mt-4 flex gap-3" role="group" aria-label={`${name} gallery`}>
   265→              {gallery.map((image, i) => (
   266→                <button
   267→                  key={`${image.src}-${i}`}
   268→                  type="button"
   269→                  onClick={() => setActiveImage(i)}
   270→                  aria-pressed={activeImage === i}
   271→                  aria-label={`View image ${i + 1} — ${image.alt}`}
   272→                  className={cn(
   273→                    'relative h-20 w-24 cursor-pointer overflow-hidden border bg-parchment transition-all duration-300 focus-visible:outline-2 focus-visible:outline-gold',
   274→                    activeImage === i
   275→                      ? 'border-gold shadow-[0_10px_24px_-14px_rgba(201,162,39,0.7)]'
   276→                      : 'border-charcoal/15 hover:border-charcoal/40',
   277→                  )}
   278→                >
   279→                  <SmartImage src={image.src} alt="" ratio="none" className="h-full w-full" />
   280→                </button>
   281→              ))}
   282→            </div>
   283→          </div>
   284→
   285→          {/* RIGHT — details */}
   286→          <div>
   287→            <p className="eyebrow text-terracotta">
   288→              {provinceLabel} · {categoryNameValue}
   289→            </p>
   290→            <h1 className="mt-4 font-display text-4xl leading-[1.06] tracking-tight text-charcoal md:text-5xl">
   291→              {name}
   292→            </h1>
   293→            {lang !== 'kh' && product.nameKh && (
   294→              <p className="mt-2 font-khmer text-base text-stone">{product.nameKh}</p>
   295→            )}
   296→
   297→            <div className="mt-5">
   298→              <button
   299→                type="button"
   300→                onClick={() =>
   301→                  document.getElementById('reviews')?.scrollIntoView({ behavior: 'smooth', block: 'start' })
   302→                }
   303→                aria-label={`${product.rating} ${t('product.reviews')} — ${t('reviews.title')}`}
   304→                className="cursor-pointer transition-opacity hover:opacity-80 focus-visible:outline-2 focus-visible:outline-gold"
   305→              >
   306→                <RatingStars
   307→                  value={product.rating}
   308→                  size="md"
   309→                  showValue
   310→                  reviews={product.reviews}
   311→                  reviewsLabel={t('product.reviews')}
   312→                />
   313→              </button>
   314→            </div>
   315→
   316→            <div className="mt-6 flex items-baseline gap-3">
   317→              <p className="font-display text-3xl tracking-tight text-charcoal">
   318→                {formatPrice(selectedSizeObj?.price ?? product.price)}
   319→              </p>
   320→              {selectedSizeObj && (
   321→                <p className="text-xs font-semibold uppercase tracking-[0.2em] text-stone">
   322→                  / {selectedSizeObj.label}
   323→                </p>
   324→              )}
   325→            </div>
   326→
   327→            <p className="mt-6 max-w-xl text-base leading-relaxed text-stone">
   328→              {product.description}
   329→            </p>
   330→
   331→            {/* Share this harvest */}
   332→            <ShareButtons title={name} className="mt-5" tone="light" />
   333→
   334→            <div className="rule mt-8" />
   335→
   336→            {/* Size */}
   337→            <div className="mt-8">
   338→              <p className="eyebrow text-stone">{t('product.size')}</p>
   339→              <div
   340→                className="mt-3 flex flex-wrap gap-2"
   341→                role="radiogroup"
   342→                aria-label={t('product.size')}
   343→              >
   344→                {sizeOptions.map((s) => {
   345→                  const active = (selectedSize || sizeOptions[0]?.label) === s.label;
   346→                  return (
   347→                    <button
   348→                      key={s.label}
   349→                      type="button"
   350→                      role="radio"
   351→                      aria-checked={active}
   352→                      onClick={() => setSelectedSize(s.label)}
   353→                      className={cn(
   354→                        'h-11 cursor-pointer border px-5 text-sm font-semibold transition-all duration-300 focus-visible:outline-2 focus-visible:outline-gold',
   355→                        active
   356→                          ? 'border-forest bg-forest text-ivory'
   357→                          : 'border-charcoal/20 text-charcoal hover:border-forest hover:text-forest',
   358→                      )}
   359→                    >
   360→                      {s.label}
   361→                    </button>
   362→                  );
   363→                })}
   364→              </div>
   365→            </div>
   366→
   367→            {/* Quantity + stock */}
   368→            <div className="mt-8">
   369→              <p className="eyebrow text-stone">{t('product.quantity')}</p>
   370→              <div className="mt-3 flex flex-wrap items-center gap-5">
   371→                <QuantityStepper value={qty} onChange={setQty} max={Math.max(product.stock, 1)} />
   372→                {product.stock > 0 ? (
   373→                  product.stock <= 20 ? (
   374→                    <span className="flex items-center gap-2.5 text-xs font-semibold uppercase tracking-[0.18em] text-terracotta">
   375→                      <span className="h-1.5 w-1.5 animate-pulse rounded-full bg-terracotta" aria-hidden="true" />
   376→                      {t('product.lowStock', { n: product.stock })}
   377→                    </span>
   378→                  ) : (
   379→                    <span className="flex items-center gap-2.5 text-xs font-semibold uppercase tracking-[0.18em] text-moss">
   380→                      <span className="h-1.5 w-1.5 rounded-full bg-moss" aria-hidden="true" />
   381→                      {t('product.inStock')} ({product.stock})
   382→                    </span>
   383→                  )
   384→                ) : (
   385→                  <span className="flex items-center gap-2.5 text-xs font-semibold uppercase tracking-[0.18em] text-terracotta">
   386→                    <span className="h-1.5 w-1.5 rounded-full bg-terracotta" aria-hidden="true" />
   387→                    {t('common.soldOut')}
   388→                  </span>
   389→                )}
   390→              </div>
   391→            </div>
   392→
   393→            {/* Actions */}
   394→            <div className="mt-10 space-y-3">
   395→              <button
   396→                type="button"
   397→                onClick={handleAdd}
   398→                disabled={product.stock <= 0}
   399→                className={cn(
   400→                  'btn-primary h-14 w-full text-xs disabled:cursor-not-allowed disabled:opacity-40',
   401→                  justAdded && 'bg-gold text-forest-deep',
   402→                )}
   403→              >
   404→                {justAdded ? (
   405→                  <>
   406→                    <Check className="h-4 w-4" strokeWidth={2} aria-hidden="true" />
   407→                    {t('common.added')}
   408→                  </>
   409→                ) : (
   410→                  t('common.addToCart')
   411→                )}
   412→              </button>
   413→              <button
   414→                type="button"
   415→                onClick={handleWishlist}
   416→                aria-pressed={wished}
   417→                className="btn-outline w-full"
   418→              >
   419→                <Heart
   420→                  className={cn('h-4 w-4', wished && 'fill-terracotta text-terracotta')}
   421→                  strokeWidth={1.5}
   422→                  aria-hidden="true"
   423→                />
   424→                {t('product.wishlist')}
   425→              </button>
   426→            </div>
   427→
   428→            {/* Meta accordion */}
   429→            <Accordion type="single" collapsible className="mt-10">
   430→              <AccordionItem value="origin" className="border-charcoal/10">
   431→                <AccordionTrigger className={accordionTrigger}>
   432→                  {t('product.origin')}
   433→                </AccordionTrigger>
   434→                <AccordionContent className="text-sm leading-relaxed text-stone">
   435→                  <p className="font-semibold text-charcoal">
   436→                    {provinceLabel}, Cambodia
   437→                  </p>
   438→                  {prov?.description && <p className="mt-2">{prov.description}</p>}
   439→                </AccordionContent>
   440→              </AccordionItem>
   441→
   442→              <AccordionItem value="farmer" className="border-charcoal/10">
   443→                <AccordionTrigger className={accordionTrigger}>
   444→                  {t('product.farmer')}
   445→                </AccordionTrigger>
   446→                <AccordionContent className="text-sm leading-relaxed text-stone">
   447→                  <p className="font-semibold text-charcoal">{product.farmerName}</p>
   448→                  {farmer && (
   449→                    <p className="mt-1 text-[11px] font-semibold uppercase tracking-[0.18em] text-stone">
   450→                      {lang === 'kh' && farmer.roleKh ? farmer.roleKh : farmer.role} ·{' '}
   451→                      {localizedProvince(farmer.province)}
   452→                    </p>
   453→                  )}
   454→                  <button
   455→                    type="button"
   456→                    onClick={goFarmer}
   457→                    disabled={!farmerSlug}
   458→                    className="mt-4 inline-flex cursor-pointer items-center gap-2 border border-charcoal/25 px-4 py-2.5 text-[10px] font-bold uppercase tracking-[0.22em] text-charcoal transition-all duration-300 hover:border-forest hover:bg-forest hover:text-ivory focus-visible:outline-2 focus-visible:outline-gold disabled:cursor-not-allowed disabled:opacity-40"
   459→                  >
   460→                    {t('product.meetFarmer')}
   461→                    <ArrowRight className="h-3 w-3" strokeWidth={1.5} aria-hidden="true" />
   462→                  </button>
   463→                </AccordionContent>
   464→              </AccordionItem>
   465→
   466→              <AccordionItem value="production" className="border-charcoal/10">
   467→                <AccordionTrigger className={accordionTrigger}>
   468→                  {t('product.production')}
   469→                </AccordionTrigger>
   470→                <AccordionContent className="text-sm leading-relaxed text-stone">
   471→                  <div className="flex flex-wrap gap-2">
   472→                    {product.organic && (
   473→                      <span className="flex items-center gap-2 border border-moss/40 bg-moss/10 px-3 py-2 text-[10px] font-bold uppercase tracking-[0.2em] text-moss">
   474→                        <Leaf className="h-3.5 w-3.5" strokeWidth={1.5} aria-hidden="true" />
   475→                        {t('common.organic')}
   476→                      </span>
   477→                    )}
   478→                    {product.sustainable && (
   479→                      <span className="flex items-center gap-2 border border-forest/30 bg-forest/5 px-3 py-2 text-[10px] font-bold uppercase tracking-[0.2em] text-forest">
   480→                        <Recycle className="h-3.5 w-3.5" strokeWidth={1.5} aria-hidden="true" />
   481→                        {t('common.sustainable')}
   482→                      </span>
   483→                    )}
   484→                    {!product.organic && !product.sustainable && (
   485→                      <p>
   486→                        {product.farmerName} · {product.unit}
   487→                      </p>
   488→                    )}
   489→                  </div>
   490→                </AccordionContent>
   491→              </AccordionItem>
   492→
   493→              <AccordionItem value="shipping" className="border-charcoal/10">
   494→                <AccordionTrigger className={accordionTrigger}>
   495→                  {t('product.shipping')}
   496→                </AccordionTrigger>
   497→                <AccordionContent className="text-sm leading-relaxed text-stone">
   498→                  <p>{t('product.shippingValue')}</p>
   499→                  <p className="mt-2 text-[11px] font-semibold uppercase tracking-[0.18em] text-stone">
   500→                    {t('announcement.shipping')}
   501→                  </p>
   502→                </AccordionContent>
   503→              </AccordionItem>
   504→            </Accordion>
   505→
   506→            {/* Farmer card */}
   507→            {farmer && (
   508→              <div className="card-editorial mt-8 flex flex-col gap-4 p-5 sm:flex-row sm:items-center">
   509→                <SmartImage
   510→                  src={farmer.portrait}
