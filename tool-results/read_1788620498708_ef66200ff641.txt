    13→import { useRecentStore } from '@/lib/stores/recent';
    14→import { useCompareStore } from '@/lib/stores/compare';

    16→import { useMounted } from '@/lib/hooks';
    17→import { useSeo } from '@/lib/useSeo';
    18→import { isSizeLow, isSizeSoldOut, sizeStock, SIZE_LOW_THRESHOLD } from '@/lib/stock';
    19→import { getCategory } from '@/lib/data/categories';

    25→import { Reveal } from '@/components/shared/Reveal';
    26→import { SectionHeading } from '@/components/shared/SectionHeading';

    30→import { ProductReviews } from '@/components/product/ProductReviews';
    31→import { SeasonLedger } from '@/components/product/SeasonLedger';
    32→import {
    33→  Accordion,

    36→  AccordionTrigger,
    37→} from '@/components/ui/accordion';

    39→
    40→// ─── ProductView — gallery, craft details, origin story & related harvests ────
    41→
    42→export default function ProductView({ view }: ViewProps) {
    43→  const { t, lang } = useLang();
    44→  const navigate = useRouterStore((s) => s.navigate);
    45→  const add = useCartStore((s) => s.add);
    46→  const toggleWishlist = useWishlistStore((s) => s.toggle);
    47→  const wishlistIds = useWishlistStore((s) => s.ids);
    48→  const recordRecent = useRecentStore((s) => s.record);
    49→  const recentSlugs = useRecentStore((s) => s.slugs);
    50→  const mounted = useMounted();
    51→
    52→  const slug = view.name === 'product' ? view.slug : '';
    53→
    54→  // ── Data ────────────────────────────────────────────────────────────────────
    55→  const {
    56→    data: product,
    57→    isError,
    58→  } = useQuery({
    59→    queryKey: ['product', slug],
    60→    queryFn: () => fetchProduct(slug),
    61→    enabled: slug.length > 0,
    62→  });
    63→  const { data: farmers } = useQuery({ queryKey: ['farmers'], queryFn: fetchFarmers });
    64→  const { data: allProducts } = useQuery({ queryKey: ['products'], queryFn: fetchProducts });
    65→
    66→  const farmer = useMemo(
    67→    () => farmers?.find((f) => f.id === product?.farmerId),
    68→    [farmers, product?.farmerId],
    69→  );
    70→  const prov = useMemo(() => (product ? getProvince(product.province) : undefined), [product]);
    71→
    72→  // ── Local state ─────────────────────────────────────────────────────────────
    73→  const [activeImage, setActiveImage] = useState(0);
    74→  const [selectedSize, setSelectedSize] = useState('');
    75→  const [qty, setQty] = useState(1);
    76→  const [justAdded, setJustAdded] = useState(false);
    77→  // Sticky mobile buy bar — appears once the main buy box scrolls away.
    78→  const [showStickyBar, setShowStickyBar] = useState(false);
    79→  // The compare tray owns the bottom edge while active — yield to it.
    80→  const trayActive = useCompareStore((s) => s.ids.length > 0);
    81→  const setBottomBarActive = useUIStore((s) => s.setBottomBarActive);
    82→  const timer = useRef<ReturnType<typeof setTimeout> | null>(null);
    83→  // Lens zoom — cursor-following magnification on fine-pointer devices.
    84→  const lensWrapRef = useRef<HTMLDivElement>(null);
    85→  // Mobile sticky add-to-cart — appears once the main buy box scrolls away.
    86→  const buyBoxRef = useRef<HTMLDivElement>(null);
    87→
    88→  // Selection state initialises from the loaded product; the keyed view wrapper remounts
    89→  // this component on navigation, so no reset effect is required.
    90→
    91→  // Clear the “Added ✓” timer on unmount.
    92→  useEffect(() => () => {
    93→    if (timer.current) clearTimeout(timer.current);
    94→  }, []);
    95→
    96→  // Track this product in the recently-viewed history — only once it has
    97→  // actually loaded, so failed/unknown slugs never pollute the history.
    98→  useEffect(() => {
    99→    if (product) recordRecent(product.slug);
   100→  }, [product?.id, recordRecent]);
   101→
   102→  // Sticky mobile buy bar — watches the main actions block; the fixed bar
   103→  // slides in only while that block is above the viewport (mobile/tablet).
   104→  useEffect(() => {
   105→    const node = buyBoxRef.current;
   106→    if (!node || typeof IntersectionObserver === 'undefined') return;
   107→    const observer = new IntersectionObserver(
   108→      ([entry]) => {
   109→        const show = !entry.isIntersecting && entry.boundingClientRect.top < 0;
   110→        setShowStickyBar(show);
   111→        setBottomBarActive(show);
   112→      },
   113→      { rootMargin: '-72px 0px 0px 0px' },
   114→    );
   115→    observer.observe(node);
   116→    return () => {
   117→      observer.disconnect();
   118→      setBottomBarActive(false);
   119→    };
   120→  }, [product, setBottomBarActive]);
   121→
   122→  // ── Gallery (product → farm → province) ─────────────────────────────────────
   123→  const gallery = useMemo(() => {
   124→    if (!product) return [];
   125→    const images: { src: string; alt: string }[] = [
   126→      { src: product.image, alt: product.name },
   127→    ];
   128→    if (farmer?.farmImage) images.push({ src: farmer.farmImage, alt: `${farmer.name} — the farm` });
   129→    if (prov) images.push({ src: prov.image, alt: `${prov.name}, Cambodia — ${prov.tagline}` });
   130→    return images;
   131→  }, [product, farmer, prov]);
   132→
   133→  // ── Related products (same category, topped up with same province) ──────────
   134→  const related = useMemo(() => {
   135→    if (!product || !allProducts) return [];
   136→    const sameCategory = allProducts.filter(
   137→      (p) => p.id !== product.id && p.category === product.category,
   138→    );
   139→    const sameProvince = allProducts.filter(
   140→      (p) =>
   141→        p.id !== product.id &&
   142→        p.category !== product.category &&
   143→        p.province === product.province,
   144→    );
   145→    return [...sameCategory, ...sameProvince].slice(0, 4);
   146→  }, [product, allProducts]);
   147→
   148→  // ── Recently viewed (history strip, excludes the current product) ───────────
   149→  const recentProducts = useMemo(() => {
   150→    if (!product || !allProducts) return [];
   151→    return recentSlugs
   152→      .filter((s) => s !== product.slug)
   153→      .map((s) => allProducts.find((p) => p.slug === s))
   154→      .filter((p): p is NonNullable<typeof p> => Boolean(p))
   155→      .slice(0, 4);
   156→  }, [product, allProducts, recentSlugs]);
   157→
   158→  // SEO — refines the base title once the product resolves (hook is called
   159→  // unconditionally to keep the hook order stable across loading states).
   160→  useSeo(
   161→    product ? `${product.name} — ${product.farmerName} | Sovann Farm` : undefined,
   162→    product
   163→      ? `${product.description} Grown in ${provinceName(product.province)}, Cambodia.`
   164→      : undefined,
   165→  );
   166→
   167→  // ── Loading / not found ─────────────────────────────────────────────────────
   168→  if (product === undefined && !isError) {
   169→    return <ProductViewSkeleton />;
   170→  }
   171→
   172→  // ── Not found ───────────────────────────────────────────────────────────────
   173→  if (!product) {
   174→    return (
   175→      <div className="container-editorial py-24">
   176→        <EmptyState
   177→          title={t('shop.empty')}
   178→          description={t('shop.emptyDesc')}
   179→          action={
   180→            <button
   181→              type="button"
   182→              onClick={() => navigate({ name: 'shop' })}
   183→              className="btn-primary"
   184→            >
   185→              {t('product.shop')}
   186→            </button>
   187→          }
   188→        />
   189→      </div>
   190→    );
   191→  }
   192→
   193→  // ── Derived content ─────────────────────────────────────────────────────────
   194→  const name = lang === 'kh' && product.nameKh ? product.nameKh : product.name;
   195→  const categoryData = getCategory(product.category);
   196→  const categoryNameValue = categoryData
   197→    ? lang === 'kh'
   198→      ? categoryData.nameKh
   199→      : categoryData.name
   200→    : null;
   201→  const provinceLabel =
   202→    product.province === 'multi'
   203→      ? lang === 'kh'
   204→        ? 'ខេត្តច្រើន'
   205→        : 'Multiple Provinces'
   206→      : lang === 'kh'
   207→        ? (prov?.nameKh ?? provinceName(product.province))
   208→        : provinceName(product.province);
   209→
   210→  const sizeOptions = product.sizes;
   211→  const selectedSizeObj =
   212→    sizeOptions.find((s) => s.label === selectedSize) ?? sizeOptions[0];
   213→  // Per-size stock: the shelf count for the size actually in hand drives the
   214→  // quantity ceiling, the stock line and the add guard.
   215→  const selectedSizeStock = selectedSizeObj
   216→    ? sizeStock(product, selectedSizeObj)
   217→    : product.stock;
   218→  const selectedSizeSoldOut = selectedSizeStock <= 0;
   219→  const activeSrc = gallery[Math.min(activeImage, Math.max(gallery.length - 1, 0))]?.src ?? product.image;
   220→  const activeAlt =
   221→    gallery[Math.min(activeImage, Math.max(gallery.length - 1, 0))]?.alt ?? product.name;
   222→
   223→  const lensMove = (e: React.MouseEvent<HTMLDivElement>) => {
   224→    if (typeof window !== 'undefined' && !window.matchMedia('(pointer: fine)').matches) return;
   225→    const wrap = lensWrapRef.current;
   226→    const target = wrap?.firstElementChild as HTMLElement | null;
   227→    if (!wrap || !target) return;
   228→    const r = wrap.getBoundingClientRect();
   229→    const x = ((e.clientX - r.left) / r.width) * 100;
   230→    const y = ((e.clientY - r.top) / r.height) * 100;
   231→    wrap.classList.add('is-lensing');
   232→    target.style.transformOrigin = `${x}% ${y}%`;
   233→    target.style.transform = 'scale(1.8)';
   234→  };
   235→
   236→  const lensLeave = () => {
   237→    const wrap = lensWrapRef.current;
   238→    const target = wrap?.firstElementChild as HTMLElement | null;
   239→    wrap?.classList.remove('is-lensing');
   240→    if (target) {
   241→      target.style.transform = 'scale(1)';
   242→      target.style.transformOrigin = 'center center';
   243→    }
   244→  };
   245→
   246→  const wished = wishlistIds.includes(product.id);
   247→  const farmerSlug =
   248→    farmer?.slug ??
   249→    (product.farmerId && product.farmerId.startsWith('f-') ? product.farmerId.slice(2) : null);
   250→  const localizedProvince = (id: ProvinceId) =>
   251→    lang === 'kh' ? (getProvince(id)?.nameKh ?? provinceName(id)) : provinceName(id);
   252→
   253→  const handleAdd = () => {
   254→    if (selectedSizeSoldOut) {
   255→      toast.error(t('product.sizeUnavailable'));
   256→      return;
   257→    }
   258→    add(product.id, selectedSizeObj?.label ?? '', qty);
   259→    setJustAdded(true);
   260→    if (timer.current) clearTimeout(timer.current);
   261→    timer.current = setTimeout(() => setJustAdded(false), 1600);
   262→    toast.success(`${name} — ${t('common.added')}`);
   263→  };
   264→
   265→  const handleWishlist = () => {
   266→    toggleWishlist(product.id);
   267→  };
   268→
   269→  const goFarmer = () => {
   270→    if (farmerSlug) navigate({ name: 'farmer', slug: farmerSlug });
   271→  };
   272→
   273→  const accordionTrigger =
   274→    'rounded-none py-5 text-[11px] font-bold uppercase tracking-[0.26em] text-charcoal hover:no-underline hover:text-forest';
   275→
   276→  return (
   277→    <div>
   278→      {/* ── Breadcrumb + back ──────────────────────────────────────────────── */}
   279→      <div className="container-editorial pt-10 md:pt-14">
   280→        <button
   281→          type="button"
   282→          onClick={() => navigate({ name: 'shop' })}
   283→          className="flex cursor-pointer items-center gap-2 text-[11px] font-bold uppercase tracking-[0.22em] text-stone transition-colors hover:text-forest focus-visible:outline-2 focus-visible:outline-gold"
   284→        >
   285→          <ArrowLeft className="h-3.5 w-3.5" strokeWidth={1.5} aria-hidden="true" />
   286→          {t('product.shop')}
   287→        </button>
   288→        <nav
   289→          aria-label="Breadcrumb"
   290→          className="mt-4 flex flex-wrap items-center gap-2 text-[11px] font-semibold uppercase tracking-[0.22em]"
   291→        >
   292→          <button
   293→            type="button"
   294→            onClick={() => navigate({ name: 'shop' })}
   295→            className="cursor-pointer text-stone transition-colors hover:text-forest focus-visible:outline-2 focus-visible:outline-gold"
   296→          >
   297→            {t('product.shop')}
   298→          </button>
   299→          {categoryNameValue && (
   300→            <>
   301→              <span className="text-charcoal/25" aria-hidden="true">/</span>
   302→              <button
   303→                type="button"
   304→                onClick={() => navigate({ name: 'shop', category: product.category })}
   305→                className="cursor-pointer text-stone transition-colors hover:text-forest focus-visible:outline-2 focus-visible:outline-gold"
   306→              >
   307→                {categoryNameValue}
   308→              </button>
   309→            </>
   310→          )}
   311→          <span className="text-charcoal/25" aria-hidden="true">/</span>
   312→          <span className="text-charcoal" aria-current="page">{name}</span>
   313→        </nav>
   314→
   315→        {/* ── Main grid ──────────────────────────────────────────────────────── */}
   316→        <div className="grid gap-12 pb-20 pt-8 lg:grid-cols-2 lg:gap-16">
   317→          {/* LEFT — gallery */}
   318→          <div>
   319→            <div
   320→              ref={lensWrapRef}
   321→              className="lens-wrap relative overflow-hidden border border-charcoal/10 bg-parchment"
   322→              onMouseMove={lensMove}
   323→              onMouseLeave={lensLeave}
   324→            >
   325→              <div className="lens-target">
   326→                <SmartImage
   327→                  key={activeSrc}
   328→                  src={activeSrc}
   329→                  alt={activeAlt}
   330→                  ratio="portrait"
   331→                  priority
   332→                />
   333→              </div>
   334→              <span
   335→                aria-hidden="true"
   336→                className="zoom-hint pointer-events-none absolute bottom-3 right-3 hidden items-center gap-1.5 bg-ivory/90 px-2.5 py-1.5 text-[9px] font-bold uppercase tracking-[0.22em] text-charcoal/70 transition-opacity duration-300 md:inline-flex"
   337→              >
   338→                <ZoomIn className="h-3 w-3" strokeWidth={1.5} />
   339→                {t('product.zoomHint')}
   340→              </span>
   341→            </div>
   342→            <div className="mt-4 flex gap-3" role="group" aria-label={`${name} gallery`}>
   343→              {gallery.map((image, i) => (
   344→                <button
   345→                  key={`${image.src}-${i}`}
   346→                  type="button"
   347→                  onClick={() => setActiveImage(i)}
   348→                  aria-pressed={activeImage === i}
   349→                  aria-label={`View image ${i + 1} — ${image.alt}`}
   350→                  className={cn(
   351→                    'relative h-20 w-24 cursor-pointer overflow-hidden border bg-parchment transition-all duration-300 focus-visible:outline-2 focus-visible:outline-gold',
   352→                    activeImage === i
   353→                      ? 'border-gold shadow-[0_10px_24px_-14px_rgba(201,162,39,0.7)]'
   354→                      : 'border-charcoal/15 hover:border-charcoal/40',
   355→                  )}
   356→                >
   357→                  <SmartImage src={image.src} alt="" ratio="none" className="h-full w-full" />
   358→                </button>
   359→              ))}
   360→            </div>
   361→          </div>
   362→
   363→          {/* RIGHT — details */}
   364→          <div>
   365→            <p className="eyebrow text-terracotta">
   366→              {provinceLabel} · {categoryNameValue}
   367→            </p>
   368→            <h1 className="mt-4 font-display text-4xl leading-[1.06] tracking-tight text-charcoal md:text-5xl">
   369→              {name}
   370→            </h1>
   371→            {lang !== 'kh' && product.nameKh && (
   372→              <p className="mt-2 font-khmer text-base text-stone">{product.nameKh}</p>
   373→            )}
   374→
   375→            <div className="mt-5">
   376→              <button
   377→                type="button"
   378→                onClick={() =>
   379→                  document.getElementById('reviews')?.scrollIntoView({ behavior: 'smooth', block: 'start' })
   380→                }
   381→                aria-label={`${product.rating} ${t('product.reviews')} — ${t('reviews.title')}`}
   382→                className="cursor-pointer transition-opacity hover:opacity-80 focus-visible:outline-2 focus-visible:outline-gold"
   383→              >
   384→                <RatingStars
   385→                  value={product.rating}
   386→                  size="md"
   387→                  showValue
   388→                  reviews={product.reviews}
   389→                  reviewsLabel={t('product.reviews')}
   390→                />
   391→              </button>
   392→            </div>
   393→
   394→            <div className="mt-6 flex items-baseline gap-3">
   395→              <p className="font-display text-3xl tracking-tight text-charcoal">
   396→                {formatPrice(selectedSizeObj?.price ?? product.price)}
   397→              </p>
   398→              {selectedSizeObj && (
   399→                <p className="text-xs font-semibold uppercase tracking-[0.2em] text-stone">
   400→                  / {selectedSizeObj.label}
   401→                </p>
   402→              )}
   403→            </div>
   404→
   405→            <p className="mt-6 max-w-xl text-base leading-relaxed text-stone">
   406→              {product.description}
   407→            </p>
   408→
   409→            {/* Harvest window — the months this harvest leaves the fields */}
   410→            <SeasonLedger product={product} className="mt-6 max-w-xl" />
   411→
   412→            {/* Share this harvest */}
   413→            <ShareButtons title={name} className="mt-5" tone="light" />
   414→
   415→            <div className="rule mt-8" />
   416→
   417→            {/* Size */}
   418→            <div className="mt-8">
   419→              <p className="eyebrow text-stone">{t('product.size')}</p>
   420→              <div
   421→                className="mt-3 flex flex-wrap gap-2"
   422→                role="radiogroup"
   423→                aria-label={t('product.size')}
   424→              >
   425→                {sizeOptions.map((s) => {
   426→                  const active = (selectedSize || sizeOptions[0]?.label) === s.label;
   427→                  const sStock = sizeStock(product, s);
   428→                  const soldOut = isSizeSoldOut(product, s);
   429→                  const low = isSizeLow(product, s);
   430→                  return (
   431→                    <button
   432→                      key={s.label}
   433→                      type="button"
   434→                      role="radio"
   435→                      aria-checked={active}
   436→                      disabled={soldOut}
   437→                      onClick={() => {
   438→                        setSelectedSize(s.label);
   439→                        // Keep the quantity honest when the shelf for this
   440→                        // size is tighter than what is already staged.
   441→                        setQty((q) => Math.min(q, Math.max(sizeStock(product, s), 1)));
   442→                      }}
   443→                      className={cn(
   444→                        'relative flex min-h-11 cursor-pointer flex-col items-center justify-center border px-4 py-1.5 transition-all duration-300 focus-visible:outline-2 focus-visible:outline-gold',
   445→                        soldOut && 'size-soldout cursor-not-allowed text-stone/60',
   446→                        !soldOut && (active
   447→                          ? 'border-forest bg-forest text-ivory'
   448→                          : 'border-charcoal/20 text-charcoal hover:border-forest hover:text-forest'),
   449→                      )}
   450→                    >
   451→                      <span className={cn('text-sm font-semibold leading-tight', soldOut && 'line-through decoration-charcoal/40')}>
   452→                        {s.label}
   453→                      </span>
   454→                      {soldOut ? (
   455→                        <span className="text-[9px] font-bold uppercase tracking-[0.14em] text-terracotta/80">
   456→                          {t('common.soldOut')}
   457→                        </span>
   458→                      ) : (
   459→                        <span
   460→                          className={cn(
   461→                            'text-[10px] font-semibold tabular-nums tracking-wide',
   462→                            active ? 'text-honey' : 'text-stone',
   463→                          )}
   464→                        >
   465→                          {formatPrice(s.price)}
   466→                          {low && (
   467→                            <span className={cn('ml-1.5', active ? 'text-honey' : 'text-terracotta')}>
   468→                              · {sStock} {t('product.sizeLeft')}
   469→                            </span>
   470→                          )}
   471→                        </span>
   472→                      )}
   473→                      {low && (
   474→                        <span
   475→                          aria-hidden="true"
   476→                          className="absolute right-1.5 top-1.5 h-1 w-1 animate-pulse rounded-full bg-terracotta"
   477→                        />
   478→                      )}
   479→                    </button>
   480→                  );
   481→                })}
   482→              </div>
   483→            </div>
   484→
   485→            {/* Quantity + stock — reflects the selected size's shelf */}
   486→            <div className="mt-8">
   487→              <p className="eyebrow text-stone">{t('product.quantity')}</p>
   488→              <div className="mt-3 flex flex-wrap items-center gap-5">
   489→                <QuantityStepper value={qty} onChange={setQty} max={Math.max(selectedSizeStock, 1)} />
   490→                {selectedSizeSoldOut ? (
   491→                  <span className="flex items-center gap-2.5 text-xs font-semibold uppercase tracking-[0.18em] text-terracotta">
   492→                    <span className="h-1.5 w-1.5 rounded-full bg-terracotta" aria-hidden="true" />
   493→                    {t('product.sizeUnavailable')}
   494→                  </span>
   495→                ) : selectedSizeStock <= SIZE_LOW_THRESHOLD ? (
   496→                  <span className="flex items-center gap-2.5 text-xs font-semibold uppercase tracking-[0.18em] text-terracotta">
   497→                    <span className="h-1.5 w-1.5 animate-pulse rounded-full bg-terracotta" aria-hidden="true" />
   498→                    {t('product.sizeOnly', { n: selectedSizeStock })}
   499→                  </span>
   500→                ) : (
   501→                  <span className="flex items-center gap-2.5 text-xs font-semibold uppercase tracking-[0.18em] text-moss">
   502→                    <span className="h-1.5 w-1.5 rounded-full bg-moss" aria-hidden="true" />
